// Clean-data reproduction of the "no logging" drift (run as a throwaway test/ file, then removed).
// Mirrors GetKcalGoalUsecase._computeDynamicCoreGoal's data selection: all weight entries ->
// fillGaps -> computeTrend; intake restricted to the 25 days before "today"; estimateTdeeSmoothed.
import 'package:flutter_test/flutter_test.dart';
import 'package:opennutritracker/core/domain/entity/weight_log_entity.dart';
import 'package:opennutritracker/core/utils/calc/tdee_dynamic_calc.dart';
import 'package:opennutritracker/core/utils/calc/weight_interpolation_calc.dart';
import 'package:opennutritracker/core/utils/calc/weight_trend_calc.dart';

void main() {
  test('dynamic TDEE while the user stops logging (clean data)', () {
    final start = DateTime(2026, 6, 29);
    final weights = [
      for (var i = 0; i <= 30; i++)
        WeightLogEntity(date: start.add(Duration(days: i)), weightKg: 85.0 - 0.15 * i),
    ];
    final intake = {for (var i = 0; i <= 30; i++) start.add(Duration(days: i)): 2300.0};
    final trend = WeightTrendCalc.computeTrend(WeightInterpolationCalc.fillGaps(weights));
    final lastLog = start.add(const Duration(days: 30));
    for (final gap in [0, 1, 3, 5, 7, 10, 14, 18, 21, 24, 25, 26]) {
      final today = lastLog.add(Duration(days: gap));
      final lookbackStart = today.subtract(const Duration(days: 25));
      final visible = {
        for (final e in intake.entries)
          if (!e.key.isBefore(lookbackStart) && !e.key.isAfter(today)) e.key: e.value,
      };
      final tdee = TdeeDynamicCalc.estimateTdeeSmoothed(trendSeries: trend, dailyIntakeKcal: visible);
      print('days_without_logging=$gap intake_days_visible=${visible.length} '
          'smoothed_tdee=${tdee?.round() ?? 'null (falls back to static)'}');
    }
  });
}
